+import { isValidDate } from './transactions.js';
+
+const kinds = new Set(['debt', 'subscription', 'project', 'custom']);
+const cadences = new Set(['once', 'weekly', 'monthly', 'yearly']);
+const statuses = new Set(['active', 'paused', 'completed', 'cancelled']);
+
+export function validateFinancialItem(input, { partial = false } = {}) {
+  if (!input || typeof input !== 'object' || Array.isArray(input)) return invalid('Les données de l’engagement sont invalides.');
+  const required = ['kind', 'name', 'plannedAmountXof', 'priority'];
+  if (!partial && required.some((key) => !(key in input))) return invalid('Complétez les champs requis de l’engagement.');
+  const result = {};
+
+  if ('kind' in input) {
+    if (!kinds.has(input.kind)) return invalid('Choisissez un type d’engagement valide.');
+    result.kind = input.kind;
+  }
+  if ('name' in input) {
+    if (typeof input.name !== 'string' || input.name.trim().length < 1 || input.name.trim().length > 120) {
+      return invalid('Le nom doit contenir de 1 à 120 caractères.');
+    }
+    result.name = input.name.trim();
+  }
+  for (const [key, output] of [
+    ['targetAmountXof', 'targetAmountXof'],
+    ['remainingAmountXof', 'remainingAmountXof'],
+    ['plannedAmountXof', 'plannedAmountXof'],
+  ]) {
+    if (!(key in input)) continue;
+    const value = input[key];
+    const minimum = key === 'targetAmountXof' ? 1 : 0;
+    if (value !== null && (!Number.isSafeInteger(value) || value < minimum)) {
+      return invalid(key === 'targetAmountXof'
+        ? 'Le montant total doit être un nombre entier positif en XOF.'
+        : 'Les montants doivent être des nombres entiers XOF positifs ou nuls.');
+    }
+    result[output] = value;
+  }
+  if ('priority' in input) {
+    if (!Number.isInteger(input.priority) || input.priority < 1 || input.priority > 5) return invalid('La priorité doit être comprise entre 1 et 5.');
+    result.priority = input.priority;
+  }
+  if ('dueOn' in input) {
+    if (input.dueOn !== null && input.dueOn !== '' && !isValidDate(input.dueOn)) return invalid('Saisissez une échéance valide.');
+    result.dueOn = input.dueOn || null;
+  }
+  if ('cadence' in input) {
+    if (input.cadence !== null && !cadences.has(input.cadence)) return invalid('Choisissez une périodicité valide.');
+    result.cadence = input.cadence;
+  }
+  if ('deferralImpact' in input) {
+    if (typeof input.deferralImpact !== 'string' || input.deferralImpact.length > 500) return invalid('La conséquence du report ne peut pas dépasser 500 caractères.');
+    result.deferralImpact = input.deferralImpact.trim();
+  }
+  if ('status' in input) {
+    if (!statuses.has(input.status)) return invalid('Choisissez un statut valide.');
+    result.status = input.status;
+  }
+  if (result.targetAmountXof != null && result.remainingAmountXof != null && result.remainingAmountXof > result.targetAmountXof) {
+    return invalid('Le solde restant ne peut pas dépasser le montant total.');
+  }
+  return { ok: true, value: result };
+}
+
+function invalid(message) {
+  return { ok: false, message };
+}
+
+export function rankObligations(items) {
+  return [...items].sort((a, b) =>
+    Number(b.priority) - Number(a.priority)
+    || (a.dueOn ?? '9999-12-31').localeCompare(b.dueOn ?? '9999-12-31')
+    || a.name.localeCompare(b.name, 'fr'),
+  );
+}

+const safeInteger = (value) => Number.isSafeInteger(Number(value));

+export function buildScenario(event, summary, items, month) {
+  if (!event || !summary || !Array.isArray(items) || !/^\d{4}-\d{2}$/.test(month)) {
+    throw new TypeError('Contexte de simulation invalide.');
+  }
+  const total = BigInt(summary.incomeXof) - BigInt(summary.expensesXof) - BigInt(summary.plannedXof);
+  if (total > BigInt(Number.MAX_SAFE_INTEGER) || total < BigInt(Number.MIN_SAFE_INTEGER)) {
+    throw new RangeError('Le solde dépasse la précision prise en charge.');
+  }
+  const availableXof = Number(total);
+  const activeItems = items.filter((item) => item.status === 'active')
+    .map((item) => ({ ...item, plannedAmountXof: Number(item.plannedAmountXof) }))
+    .filter((item) => Number.isSafeInteger(item.plannedAmountXof) && item.plannedAmountXof > 0);
+
+  const candidateActions = activeItems.flatMap((item) => {
+    const release = Math.min(item.plannedAmountXof, Number(event.amountXof));
+    const actions = [{
+      itemId: item.id,
+      itemName: item.name,
+      action: 'reduce',
+      amountDeltaXof: release,
+      consequence: item.deferralImpact || 'Réduire le montant prévu.',
+      itemVersion: Number(item.version),
+      priority: Number(item.priority),
+      dueOn: item.dueOn,
+    }];
+    if (item.dueOn || item.cadence === 'once') {
+      actions.push({
+        itemId: item.id,
+        itemName: item.name,
+        action: 'defer',
+        amountDeltaXof: item.plannedAmountXof,
+        consequence: item.deferralImpact || 'Reporter la prochaine échéance.',
+        itemVersion: Number(item.version),
+        priority: Number(item.priority),
+        dueOn: item.dueOn,
+      });
+    }
+    actions.push({
+      itemId: item.id,
+      itemName: item.name,
+      action: 'pause',
+      amountDeltaXof: item.plannedAmountXof,
+      consequence: item.deferralImpact || 'Mettre temporairement en pause cet engagement.',
+      itemVersion: Number(item.version),
+      priority: Number(item.priority),
+      dueOn: item.dueOn,
+    });
+    return actions;
+  });
+
+  const baselineAfter = availableXof - Number(event.amountXof);
+  if (!Number.isSafeInteger(baselineAfter)) throw new RangeError('Le solde dépasse la précision prise en charge.');
+  const candidates = [{
+    id: 'use-available-balance',
+    actions: [{
+      itemId: null,
+      itemName: 'Solde disponible',
+      action: 'allocate',
+      amountDeltaXof: Number(event.amountXof),
+      consequence: 'Affecter le solde disponible à cet imprévu.',
+      itemVersion: 0,
+      priority: 5,
+      dueOn: event.dueOn ?? null,
+    }],
+    availableBeforeXof: availableXof,
+    eventAmountXof: Number(event.amountXof),
+    availableAfterXof: baselineAfter,
+    explanation: 'Financer l’imprévu avec le solde disponible, sans modifier les engagements.',
+    priorityScore: 0,
+    admissible: baselineAfter >= 0,
+  }];
+  for (const action of candidateActions) {
+    candidates.push(makeCandidate([action], availableXof, Number(event.amountXof)));
+  }
+  const orderedActions = [...candidateActions].sort((a, b) =>
+    b.amountDeltaXof - a.amountDeltaXof || a.itemId.localeCompare(b.itemId) || a.action.localeCompare(b.action),
+  );
+  for (let first = 0; first < orderedActions.length && candidates.length < 48; first += 1) {
+    for (let second = first + 1; second < orderedActions.length && candidates.length < 48; second += 1) {
+      if (orderedActions[first].itemId === orderedActions[second].itemId) continue;
+      candidates.push(makeCandidate([orderedActions[first], orderedActions[second]], availableXof, Number(event.amountXof)));
+    }
+  }
+  const unique = [...new Map(candidates.map((candidate) => [candidate.id, candidate])).values()];
+  const ranked = unique.sort((a, b) =>
+    (b.availableAfterXof >= 0) - (a.availableAfterXof >= 0)
+    || b.priorityScore - a.priorityScore
+    || a.actions.length - b.actions.length
+    || a.id.localeCompare(b.id),
+  );
+  const selected = ranked.slice(0, 12);
+  if (!selected.some((candidate) => candidate.id === 'use-available-balance')) selected[selected.length - 1] = candidates[0];
+
+  const dataVersion = createHash('sha256').update(JSON.stringify({
+    event: { id: event.id, version: Number(event.version), amountXof: Number(event.amountXof) },

+    summary: [summary.incomeXof, summary.expensesXof, summary.plannedXof],
+    items: activeItems.map(({ id, version, plannedAmountXof, status }) => [id, Number(version), plannedAmountXof, status]).sort(),
+  })).digest('hex');
+  return {
+    dataVersion,
+    availableBeforeXof: availableXof,
+    event: { id: event.id, title: event.title, amountXof: Number(event.amountXof), urgency: Number(event.urgency) },
+    candidates: selected,
+  };
+}

+function makeCandidate(actions, available, eventAmount) {
+  const freed = actions.reduce((sum, action) => sum + (action.action === 'allocate' ? 0 : action.amountDeltaXof), 0);
+  if (!safeInteger(freed) || !safeInteger(available + freed - eventAmount)) throw new RangeError('Le solde dépasse la précision prise en charge.');
+  const normalized = [...actions].sort((a, b) => a.itemId.localeCompare(b.itemId) || a.action.localeCompare(b.action));
+  const id = createHash('sha256').update(JSON.stringify(normalized.map(({ itemId, action, amountDeltaXof, itemVersion }) => [itemId, action, amountDeltaXof, itemVersion]))).digest('hex').slice(0, 24);
+  return {
+    id,
+    actions: normalized,
+    availableBeforeXof: available,
+    eventAmountXof: eventAmount,
+    availableAfterXof: available + freed - eventAmount,
+    explanation: normalized.map((action) => `${action.action}: ${action.consequence}`).join(' '),
+    priorityScore: normalized.reduce((score, action) => score + (6 - action.priority) * 1000, 0),
+    admissible: available + freed - eventAmount >= 0,
+  };
+}

+import { createHash } from 'node:crypto';
+
+export function deterministicRecommendations(candidates) {
+  if (!Array.isArray(candidates)) throw new TypeError('Les options de recommandation sont invalides.');
+  const ranked = [...candidates].sort((a, b) =>
+    (b.availableAfterXof >= 0) - (a.availableAfterXof >= 0)
+    || a.actions.length - b.actions.length
+    || b.availableAfterXof - a.availableAfterXof
+    || a.id.localeCompare(b.id),
+  );
+  return ranked.map((candidate, index) => ({
+    candidateId: candidate.id,
+    rank: index + 1,
+    explanation: candidate.explanation,
+    availableAfterXof: candidate.availableAfterXof,
+  }));
+}
+
+export function candidateHash(candidates) {
+  return createHash('sha256').update(JSON.stringify(candidates.map(({ id, actions, availableAfterXof }) => [id, actions, availableAfterXof]))).digest('hex');
+}
+
+export function validateGroqRanking(payload, candidateIds) {
+  if (!payload || Object.keys(payload).length !== 1 || !Array.isArray(payload.options) || payload.options.length !== candidateIds.length) return null;
+  const permitted = new Set(candidateIds);
+  const seen = new Set();
+  const result = [];
+  for (const option of payload.options) {
+    if (!option || Object.keys(option).some((key) => !['candidateId', 'explanation'].includes(key))
+        || Object.keys(option).length !== 2) return null;
+    if (!option || typeof option.candidateId !== 'string' || !permitted.has(option.candidateId) || seen.has(option.candidateId)) return null;
+    if (typeof option.explanation !== 'string' || option.explanation.trim().length < 1 || option.explanation.length > 240) return null;
+    seen.add(option.candidateId);
+    result.push({ candidateId: option.candidateId, explanation: option.explanation.trim() });
+  }
+  return seen.size === permitted.size ? result : null;
+}

+      SELECT
+        COALESCE(SUM(amount_xof) FILTER (WHERE kind = 'income' AND status = 'paid'), 0)::bigint AS "incomeXof",
+        COALESCE(SUM(amount_xof) FILTER (WHERE kind = 'expense' AND status = 'paid'), 0)::bigint AS "expensesXof",
+        COALESCE(SUM(amount_xof) FILTER (WHERE kind = 'expense' AND status <> 'paid'), 0)::bigint
+        + COALESCE((
+          SELECT SUM(GREATEST(0, fi.planned_amount_xof - COALESCE(paid.amount_xof, 0)))
+          FROM financial_item fi
+          LEFT JOIN LATERAL (
+            SELECT SUM(t.amount_xof)::bigint AS amount_xof

+      WHERE owner_id = ${ownerId}
+        AND (CASE WHEN status = 'paid' THEN occurred_on ELSE COALESCE(due_on, occurred_on) END) >= ${firstDay}::date
+        AND (CASE WHEN status = 'paid' THEN occurred_on ELSE COALESCE(due_on, occurred_on) END) < ${nextMonth}::date
+    `,
+    sql`
+      SELECT fi.id, fi.name,
+        GREATEST(0, fi.planned_amount_xof - COALESCE(paid.amount_xof, 0))::bigint AS "plannedAmountXof",
+        fi.due_on::text AS "dueOn", fi.cadence, fi.priority, fi.deferral_impact AS "deferralImpact",
+        fi.status, fi.version
+      FROM financial_item fi

+    payload = JSON.parse(responseBody) as T | ApiError;

+const CACHE = 'budget-shell-v1';
+const SHELL = ['/', '/index.html', '/manifest.webmanifest', '/icon.svg'];
+
+self.addEventListener('install', (event) => {
+  event.waitUntil(caches.open(CACHE).then((cache) => cache.addAll(SHELL)));
+  self.skipWaiting();
+});
+
+self.addEventListener('activate', (event) => {
+  event.waitUntil(
+    caches.keys().then((keys) => Promise.all(keys.filter((key) => key !== CACHE).map((key) => caches.delete(key)))),
+  );
+  self.clients.claim();
+});
+
+self.addEventListener('fetch', (event) => {
+  const request = event.request;
+  const url = new URL(request.url);
+  if (request.method !== 'GET' || url.origin !== self.location.origin) return;
+  if (url.pathname === '/api' || url.pathname.startsWith('/api/')) {
+    event.respondWith(fetch(request));
+    return;
+  }
+  if (request.mode === 'navigate') {
+    event.respondWith(fetch(request).catch(() => caches.match('/')));
+    return;
+  }
+  if (url.pathname.startsWith('/assets/')) {
+    event.respondWith(caches.open(CACHE).then(async (cache) => {
+      const cached = await cache.match(request);
+      if (cached) return cached;
+      const response = await fetch(request);
+      if (response.ok && response.type === 'basic') await cache.put(request, response.clone());
+      return response;
+    }));
+  }
+});

+# Budget
+
+Application personnelle de suivi mensuel des revenus et dépenses en XOF, conçue pour le téléphone et l’ordinateur.
+
+## Développement local
+
+Prérequis : Node.js 24 LTS et npm.
+
+```powershell
+npm install
+npm run dev:all
+```
+
+Ouvrir l’URL Vite affichée dans le terminal. L’API locale écoute sur `http://127.0.0.1:3000` et le frontend relaie `/api` vers cette API.
+
+## PostgreSQL et premier compte
+
+1. Créer une base PostgreSQL managée (Neon via Vercel Marketplace est prévu).
+2. Appliquer les migrations dans l’ordre : `db/migrations/001_initial.sql`, puis `db/migrations/002_financial_planning.sql`. Ne pas réexécuter une migration déjà appliquée.
+3. Dans une fenêtre PowerShell, fournir l’URL de connexion sans l’ajouter au dépôt :
+
+   ```powershell
+   $env:DATABASE_URL = "postgresql://..."
+   npm run owner:create
+   ```
+
+4. Saisir l’adresse courriel et un nouveau mot de passe lorsqu’ils sont demandés. La commande refuse de créer un second compte et n’affiche ni n’écrit le mot de passe.
+5. Pour effacer la variable de la session PowerShell après la création :
+
+   ```powershell
+   Remove-Item Env:DATABASE_URL
+   ```
+
+Le compte propriétaire est créé uniquement par cette commande d’administration. La page de connexion ne permet pas une inscription publique. Sans `DATABASE_URL`, les tests et le build restent exécutables, mais les parcours authentifiés nécessitent une base configurée.
+
+## Engagements et arbitrages
+
+- Les dettes, abonnements, projets et postes personnalisés sont des engagements planifiés distincts des mouvements réellement payés. Une dépense peut être liée à un engagement; les paiements liés sont déduits des montants planifiés pour éviter un double comptage.
+- Les imprévus se simulent sans modifier les données financières. Les options non couvrables affichent le déficit et ne peuvent pas être confirmées. La confirmation verrouille et vérifie les versions, enregistre le scénario et ses actions dans une seule instruction atomique et exige une clé idempotente.
+- Groq est facultatif. Définir `GROQ_API_KEY` uniquement dans l’environnement serveur. La demande n’envoie ni note, ni bénéficiaire, ni historique brut, ni identifiant d’engagement; les règles déterministes restent disponibles si Groq est désactivé, limité, indisponible ou renvoie une réponse invalide.
+- L’export JSON est privé et exige une confirmation dans l’interface. Aucun endpoint de suppression automatique du compte n’est proposé; une demande est traitée manuellement avant la mise en production.
+- La PWA ne met en cache que le shell et les ressources statiques. Les routes `/api`, sessions, exports et données financières ne sont jamais mises en cache. Les mutations sont désactivées hors connexion.
+
+## Vérifications
+
+```powershell
+npm run build
+npm test
+npm audit --omit=dev --audit-level=high
+```
+
+## Déploiement Vercel
+
+- Connecter le dépôt à Vercel et provisionner PostgreSQL via Marketplace.
+- Configurer les valeurs de connexion fournies par le fournisseur uniquement dans les variables serveur Vercel.
+- Appliquer les migrations avant d’exposer l’application et lancer `npm run owner:create` une seule fois depuis un terminal privé connecté à la base de production choisie.
+- Déployer une Preview privée avant Production et utiliser une base distincte pour Preview.
+- La seule variable propre à Groq est `GROQ_API_KEY`; elle ne sera nécessaire qu’à l’intégration des recommandations, pas pour le suivi des transactions.
+
+Ne jamais placer de mot de passe, d’URL de base réelle ou de clé Groq dans Git, les fichiers de migration ou le bundle navigateur. Les montants sont des francs CFA XOF entiers et les mois sont interprétés selon `Africa/Abidjan` (GMT+0).
